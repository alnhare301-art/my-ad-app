import express from "express"; import path from "path"; import { createServer } from "http"; import { Server } from "socket.io"; import { createServer as createViteServer } from "vite"; const PORT = 3000; async function startServer() { const app = express(); const httpServer = createServer(app); const io = new Server(httpServer, { cors: { origin: "*", methods: ["GET", "POST"] }, }); // حالة اللعبة interface Player { id: string; name: string; avatar: string; amount: number; color: string; } let playersMap: Map = new Map(); let gameState: "waiting" | "counting" | "playing" | "winner" = "waiting"; let countdown = 10; let timerInterval: NodeJS.Timeout | null = null; let gameSeed = 0; let winner: Player | null = null; let pot = 0; const COLORS = ["#3b82f6", "#ef4444", "#10b981", "#a855f7", "#f97316", "#eab308", "#06b6d4"]; // دالة تصفير الساحة بالكامل function resetGame() { console.log("♻️ جاري تصفير الساحة لجولة جديدة..."); playersMap.clear(); gameState = "waiting"; countdown = 10; gameSeed = 0; winner = null; pot = 0; if (timerInterval) clearInterval(timerInterval); timerInterval = null; // إرسال تحديث التصفير لجميع المتصلين io.emit("gameUpdate", { gameState, players: [], countdown, winner: null, pot: 0 }); } function startGame() { gameState = "playing"; gameSeed = Math.random(); io.emit("gameStart", { seed: gameSeed, players: Array.from(playersMap.values()) }); // وقت حركة الكرة (12 ثانية) setTimeout(() => { calculateWinner(); }, 12000); } function calculateWinner() { if (playersMap.size === 0) { resetGame(); return; } const playerList = Array.from(playersMap.values()); const totalAmount = playerList.reduce((sum, p) => sum + p.amount, 0); let random = Math.random() * totalAmount; let current = 0; for (const p of playerList) { current += p.amount; if (random <= current) { winner = p; break; } } if (!winner) winner = playerList[0]; gameState = "winner"; io.emit("gameWinner", { winner, pot: totalAmount }); // عرض شاشة الفوز لمدة 8 ثوانٍ ثم التصفير التلقائي setTimeout(() => { resetGame(); }, 8000); } io.on("connection", (socket) => { // إرسال الحالة الحالية للمستخدم الجديد socket.emit("gameUpdate", { gameState, players: Array.from(playersMap.values()), countdown, winner, pot }); socket.on("join", (playerData: { id: string; name: string; avatar: string; amount: number }) => { if (gameState !== "waiting" && gameState !== "counting") return; const existingPlayer = playersMap.get(playerData.id); if (existingPlayer) { existingPlayer.amount += playerData.amount; } else { playersMap.set(playerData.id, { ...playerData, color: COLORS[playersMap.size % COLORS.length] }); } io.emit("gameUpdate", { gameState, players: Array.from(playersMap.values()), countdown }); // بدء العد التنازلي عند رصد لاعبين if (playersMap.size >= 2 && gameState === "waiting") { gameState = "counting"; countdown = 10; timerInterval = setInterval(() => { countdown--; io.emit("countdown", countdown); if (countdown <= 0) { clearInterval(timerInterval!); timerInterval = null; startGame(); } }, 1000); } }); // إضافة ميزة التصفير اليدوي للمطور (اختياري) socket.on("forceReset", () => { resetGame(); }); }); // تكامل Vite للتشغيل if (process.env.NODE_ENV !== "production") { const vite = await createViteServer({ server: { middlewareMode: true }, appType: "spa" }); app.use(vite.middlewares); } else { const distPath = path.join(process.cwd(), "dist"); app.use(express.static(distPath)); app.get("*", (req, res) => res.sendFile(path.join(distPath, "index.html"))); } httpServer.listen(PORT, "0.0.0.0", () => { console.log(`Server running on port ${PORT}`); }); } startServer();
